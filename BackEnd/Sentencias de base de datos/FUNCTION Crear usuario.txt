DROP FUNCTION usuario_crear(text,text,text,text,text,text,character,character,date);
CREATE OR REPLACE FUNCTION usuario_crear(
    p_identifi    		text,
    p_nombre           	text,
    p_apellidos        	text,
    p_email            	text,
    p_clave   			text,
    p_telefono         	text DEFAULT NULL,
    p_genero           	char(2) DEFAULT NULL,
    p_pais             	char(2) DEFAULT NULL,
    p_fec_naci 			date DEFAULT NULL
)
RETURNS VARCHAR(100)
LANGUAGE plpgsql
AS $$
DECLARE
  v_id integer;
BEGIN
  -- Validar existencia previa por identificador o email
  IF EXISTS (
    SELECT 1 FROM usuarios 
    WHERE identifi = p_identifi OR email = p_email
  ) THEN
    RETURN 'Correo o documento ya existe';
  END IF;

  -- Insertar nuevo registro con contraseña encriptada
  INSERT INTO usuarios (
    identifi, nombre, apellidos, email, clave, 
    telefono, genero, pais, fec_naci, creado_en, last_update
  )
  VALUES (
    p_identifi,
    p_nombre,
    p_apellidos,
    p_email,
    crypt(p_clave, gen_salt('bf')), -- Cifrado con bcrypt
    p_telefono,
    p_genero,
    p_pais,
    p_fec_naci,
    NOW(),
    NOW()
  )
  RETURNING id INTO v_id;

  RETURN v_id;
END;
$$;

-- Requerido para el uso de crypt() y gen_salt()
CREATE EXTENSION IF NOT EXISTS pgcrypto;
