CREATE OR REPLACE VIEW public.vista_historial_tickets AS
 SELECT r.id AS id_sorteo,
    t.id_usuario,
    t.codigo_ticket,
    r.title,
    t.estado_pago,
    t.monto_total,
    t.metodo_pago,
    t.fecha_compra,
    t.imagen_comprobante,
	u.nombre,
	u.apellidos,
	u.telefono,
	u.email,
	u.identifi,
	t.observaciones
   FROM tickets_sorteos t
     LEFT JOIN raffles r ON r.id::text = t.id_sorteo::text
	 LEFT JOIN usuarios u ON u.id = t.id_usuario
 WHERE t.estado_pago::text = 'pendiente'::text;

ALTER TABLE public.vista_historial_tickets OWNER TO postgres;
