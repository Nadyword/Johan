CREATE TABLE tickets_sorteos (
    id_ticket SERIAL PRIMARY KEY,
    id_sorteo VARCHAR(50) NOT NULL REFERENCES raffles(id) ON DELETE CASCADE,
    id_usuario INT NOT NULL REFERENCES usuarios(id) ON DELETE CASCADE,
    codigo_ticket VARCHAR(50) UNIQUE NOT NULL,
    cantidad_tickets INT NOT NULL DEFAULT 1,
    monto_unitario NUMERIC(12,2) NOT NULL,
    monto_total NUMERIC(12,2) GENERATED ALWAYS AS (cantidad_tickets * monto_unitario) STORED,
    metodo_pago VARCHAR(30) NOT NULL,
    estado_pago VARCHAR(20) NOT NULL DEFAULT 'pendiente' 
        CHECK (estado_pago IN ('pendiente', 'pagado', 'cancelado')),
    imagen_comprobante TEXT, -- URL o base64 del comprobante si aplica
    fecha_compra TIMESTAMP NOT NULL DEFAULT NOW(),
    fecha_actualizacion TIMESTAMP DEFAULT NOW(),
    imagen_sorteo  VARCHAR(300) NOT NULL,
    observaciones TEXT
);
