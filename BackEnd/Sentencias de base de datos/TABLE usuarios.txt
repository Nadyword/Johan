CREATE TABLE usuarios (
    id 			                SERIAL PRIMARY KEY,                         -- Identificador único
	identifi	                VARCHAR(50) NOT NULL,			            -- Documento de identificacion
    nombre 		        VARCHAR(200) NOT NULL,         	        -- Nombre del usuario
	apellidos 	            VARCHAR(200) NOT NULL, 			        -- Apellido del usuario
    email 		            VARCHAR(250) UNIQUE NOT NULL,   -- Correo electrónico (para login/contacto)
	genero		            CHAR(2),						                            -- Genero 
	pais		                CHAR(2),						                            -- Pais de Origen
    clave	 	            TEXT NOT NULL,               	                    -- Contraseña (idealmente encriptada con bcrypt/argon2)
    telefono 	            VARCHAR(25),                 	                    -- Teléfono opcional
    fec_naci 	            DATE,                			                            -- Fecha de nacimiento
    creado_en 	        TIMESTAMP DEFAULT NOW(),    	        -- Fecha de creación
    last_update         TIMESTAMP DEFAULT NOW() 		        -- Última actualización
);
