CREATE OR REPLACE VIEW public.vista_tickets_sorteos AS
 SELECT r.title AS titulo_sorteo,
    t.codigo_ticket,
    t.fecha_compra,
    r.ends_at AS fecha_finalizacion,
    r.status AS sorteo_activo,
    t.estado_pago,
    t.id_sorteo,
    t.id_usuario,
    t.imagen_sorteo,
	itn.*
   FROM tickets_sorteos t
   LEFT JOIN raffles r ON t.id_sorteo::text = r.id::text
   LEFT JOIN info_tickets_numeros itn ON t.codigo_ticket::int = itn.id_ticket
   ORDER BY r.status DESC;

ALTER TABLE public.vista_tickets_sorteos OWNER TO postgres;
